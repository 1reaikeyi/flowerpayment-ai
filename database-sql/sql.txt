create table chat_record
(
    id              bigint                             not null comment '数据id'
        primary key,
    conversation_id varchar(100) collate utf8mb4_bin   not null comment '对话id',
    data            text collate utf8mb4_bin           null comment '对话数据',
    create_time     datetime default CURRENT_TIMESTAMP not null comment '创建时间',
    update_time     datetime default CURRENT_TIMESTAMP not null on update CURRENT_TIMESTAMP comment '更新时间'
)
    comment '对话记录';

create index conversation_id_index
    on chat_record (conversation_id);

create index create_time_index
    on chat_record (create_time);

create table chat_session
(
    id          bigint                             not null comment '数据id'
        primary key,
    session_id  varchar(32)                        not null comment '会话id',
    user_id     bigint   default 0                 not null comment '用户id',
    title       varchar(100) collate utf8mb4_bin   null comment '会话标题',
    create_time datetime default CURRENT_TIMESTAMP not null comment '创建时间',
    update_time datetime default CURRENT_TIMESTAMP not null on update CURRENT_TIMESTAMP comment '更新时间'
)
    comment 'session';

create index session_id_index
    on chat_session (session_id);

create index update_time_index
    on chat_session (update_time);

create index user_id_index
    on chat_session (user_id);

create table employee
(
    id          bigint auto_increment comment '主键'
        primary key,
    work        varchar(32)      not null comment '工作',
    username    varchar(32)      not null comment '登录账号',
    password    varchar(64)      not null comment '密码',
    avatar      varchar(64)      null comment '头像',
    email       varchar(32)      not null comment '邮箱',
    phone       varchar(11)      not null comment '手机号',
    sex         varchar(2)       not null comment '性别',
    status      bigint default 1 not null comment '状态 0:禁用，1:启用',
    create_time datetime         null comment '创建时间',
    update_time datetime         null comment '更新时间',
    create_user bigint           null comment '创建人',
    update_user bigint           null comment '修改人',
    constraint idx_username
        unique (username)
)
    comment '员工信息（店长/店员/客服）' collate = utf8mb4_bin;

create table festival
(
    id          bigint auto_increment comment '主键'
        primary key,
    category_id bigint           null comment '所属分类',
    name        varchar(32)      not null comment '多花组合名字',
    price       decimal(10, 2)   not null comment '价格',
    number      bigint           not null comment '鲜花数量',
    status      bigint default 1 null comment '售卖状态 0:下架 1:在售',
    description varchar(255)     null comment '礼盒描述',
    image       varchar(255)     null comment '礼盒图片',
    create_time datetime         null comment '创建时间',
    update_time datetime         null comment '更新时间',
    create_user bigint           null comment '创建人',
    update_user bigint           null comment '修改人',
    constraint idx_festival_name
        unique (name)
)
    comment '多花组合' collate = utf8mb4_bin;

create table festival_detail
(
    id          bigint auto_increment comment '主键'
        primary key,
    festival_id bigint       null comment '多花组合id',
    flower_id   bigint       null comment '关联鲜花id',
    spec_object varchar(32)  null comment '送人对象',
    spec_option varchar(255) null comment '用途'
)
    comment '多花详细信息' collate = utf8mb4_bin;

create index idx_festival_id
    on festival_detail (festival_id)
    comment 'festival的索引';

create index idx_flower_id
    on festival_detail (flower_id)
    comment 'flower的index';
